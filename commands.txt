================================================================================
  CART ABANDONMENT PIPELINE — FULL RUN COMMANDS
  Run top to bottom. Each section assumes the previous succeeded.

────────────────────────────────────────────────────────────────────────────────
0. PRE-FLIGHT — HOST SETUP
────────────────────────────────────────────────────────────────────────────────

# Install Python dependencies (host only — producer runs on host)
pip install -r requirements.txt

# Copy your GCP service account key into credentials/
cp /path/to/your/gcp-key.json credentials/gcp-key.json

# Verify .env is populated
cat .env

────────────────────────────────────────────────────────────────────────────────
1. TERRAFORM — PROVISION GCS + BIGQUERY
────────────────────────────────────────────────────────────────────────────────

cd infra
terraform init
terraform plan
terraform apply -auto-approve
cd ..

────────────────────────────────────────────────────────────────────────────────
2. DOCKER — START ALL CONTAINERS
────────────────────────────────────────────────────────────────────────────────

docker compose up -d

# Verify all 4 containers are running
docker compose ps

# Wait for Kafka to be healthy (up to ~60s)
docker compose logs kafka --follow
# Ctrl-C once you see: "Kafka Server started"

────────────────────────────────────────────────────────────────────────────────
3. KAFKA — CREATE TOPIC
────────────────────────────────────────────────────────────────────────────────

docker exec kafka kafka-topics.sh \
  --bootstrap-server localhost:9092 \
  --create \
  --topic cart-events \
  --partitions 3 \
  --replication-factor 1

# Verify topic exists
docker exec kafka kafka-topics.sh \
  --bootstrap-server localhost:9092 \
  --list

# Kafka UI (optional, browser)
# http://localhost:8090

────────────────────────────────────────────────────────────────────────────────
4. PRODUCER — START PUBLISHING EVENTS (HOST)
────────────────────────────────────────────────────────────────────────────────

# Run in its own terminal — leave it running
python producer/cart_producer.py

# Expected output:
# [producer] Connected to localhost:29092, publishing to 'cart-events'
# [session     1] page_view            user=...  product=...

────────────────────────────────────────────────────────────────────────────────
5. BRONZE — START STREAMING JOB (runs until you Ctrl-C)
────────────────────────────────────────────────────────────────────────────────

# Run in its own terminal — leave it running alongside the producer
docker exec spark-master /opt/spark/bin/spark-submit \
  --master spark://spark-master:7077 \
  --packages io.delta:delta-spark_2.12:3.2.0,org.apache.spark:spark-sql-kafka-0-10_2.12:3.5.1,com.google.cloud.bigdataoss:gcs-connector:hadoop3-2.2.19,com.google.cloud.spark:spark-bigquery-with-dependencies_2.12:0.36.1 \
  --conf spark.jars.ivy=/tmp/.ivy \
  --conf spark.sql.extensions=io.delta.sql.DeltaSparkSessionExtension \
  --conf spark.sql.catalog.spark_catalog=org.apache.spark.sql.delta.catalog.DeltaCatalog \
  --conf spark.hadoop.fs.gs.impl=com.google.cloud.hadoop.fs.gcs.GoogleHadoopFileSystem \
  --conf spark.hadoop.fs.AbstractFileSystem.gs.impl=com.google.cloud.hadoop.fs.gcs.GoogleHadoopFS \
  --conf spark.hadoop.google.cloud.auth.service.account.json.keyfile=/opt/credentials/gcp-key.json \
  /opt/spark-apps/cart_bronze.py

# Expected output:
# [bronze] Streaming to gs://cart-pipeline-lake/bronze/cart_events — waiting for data...
# Spark UI: http://localhost:8080

# Let bronze run for at least 2-3 minutes to accumulate data before running silver.

────────────────────────────────────────────────────────────────────────────────
9. VERIFY — BIGQUERY
────────────────────────────────────────────────────────────────────────────────

# Row counts
bq query --use_legacy_sql=false \
  'SELECT COUNT(*) AS total FROM cart_pipeline.abandoned_carts'

bq query --use_legacy_sql=false \
  'SELECT COUNT(*) AS total FROM cart_pipeline.hourly_stats'

# Top abandoned carts by value
bq query --use_legacy_sql=false \
  'SELECT session_id, user_id, cart_value, minutes_inactive, abandoned_at
   FROM cart_pipeline.abandoned_carts
   ORDER BY cart_value DESC
   LIMIT 10'

# Hourly abandonment breakdown
bq query --use_legacy_sql=false \
  'SELECT hour, total_sessions, abandoned_sessions, abandonment_rate, revenue_lost
   FROM cart_pipeline.hourly_stats
   ORDER BY hour'
